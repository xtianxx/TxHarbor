import collections
import json
import sys

base = sys.argv[1] if len(sys.argv) > 1 else "docs/evidence/017-carrier-identity-r1-strict-classifier/logs"


def scan(path):
    actions = collections.Counter()
    subtests = collections.defaultdict(set)
    tops = []
    with open(path) as fh:
        for line in fh:
            line = line.strip()
            if not line:
                continue
            ev = json.loads(line)
            action = ev.get("Action")
            actions[action] += 1
            name = ev.get("Test")
            if not name:
                continue
            if "/" in name:
                subtests[name.split("/", 1)[0]].add(name)
            elif action == "run":
                tops.append(name)
    return actions, tops, subtests


for fname in (
    "identity-unit.go-test.json",
    "identity-unit-race.go-test.json",
    "carrier-real-docker.go-test.json",
):
    actions, tops, subtests = scan(f"{base}/{fname}")
    print(f"== {fname} ==")
    print(f"   events: pass={actions['pass']} fail={actions['fail']} skip={actions['skip']}")
    print(f"   top-level tests: {len(tops)}")
    for top in tops:
        print(f"     - {top}: subtests={len(subtests.get(top, ()))}")
    print()
PY
